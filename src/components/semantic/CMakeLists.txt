cmake_minimum_required(VERSION 3.24)
find_package(Qt6 6.11.1 REQUIRED COMPONENTS Core Sql)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Pinned ONNX Runtime CPU package root")
set(HNSWLIB_ROOT "" CACHE PATH "Pinned hnswlib source/package root")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
find_path(HNSWLIB_INCLUDE_DIR hnswlib/hnswlib.h HINTS "${HNSWLIB_ROOT}" "${HNSWLIB_ROOT}/include" REQUIRED)
qt_add_executable(ella-semantic main.cpp)
target_include_directories(ella-semantic PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}" "${HNSWLIB_INCLUDE_DIR}")
target_link_libraries(ella-semantic PRIVATE Qt6::Core Qt6::Sql "${ONNXRUNTIME_LIBRARY}")
set_target_properties(ella-semantic PROPERTIES WIN32_EXECUTABLE FALSE INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
install(TARGETS ella-semantic RUNTIME DESTINATION bin)
